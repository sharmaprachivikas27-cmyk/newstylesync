<script>
const scenes = document.querySelectorAll(".scene");
let sceneIndex = 0;
let playing = false;
let sceneStart = 0;
let elapsed = 0;
let timer = null;

const durations = [15, 20, 30, 20, 20, 35, 30, 10];
const totalDuration = 180;

function showScene(index) {
  scenes.forEach((scene, i) => {
    scene.classList.toggle("active", i === index);
  });
  sceneIndex = index;
}

function updateTimer() {
  const seconds = Math.min(
    totalDuration,
    elapsed + (playing ? (Date.now() - sceneStart) / 1000 : 0)
  );

  const m = Math.floor(seconds / 60);
  const s = Math.floor(seconds % 60);

  document.getElementById("timer").textContent =
    String(m).padStart(2, "0") + ":" +
    String(s).padStart(2, "0") + " / 03:00";

  if (seconds >= totalDuration) {
    pauseVideo();
    elapsed = totalDuration;
  }
}

function tick() {
  if (!playing) return;

  const currentTime =
    elapsed + (Date.now() - sceneStart) / 1000;

  let cumulative = 0;

  for (let i = 0; i < durations.length; i++) {
    cumulative += durations[i];

    if (currentTime < cumulative) {
      showScene(i);
      break;
    }
  }

  updateTimer();
  timer = requestAnimationFrame(tick);
}

function startVideo() {
  if (playing) return;

  if (elapsed >= totalDuration) {
    elapsed = 0;
    showScene(0);
  }

  playing = true;
  sceneStart = Date.now();
  tick();
}

function pauseVideo() {
  if (!playing) return;

  elapsed += (Date.now() - sceneStart) / 1000;
  playing = false;

  if (timer) cancelAnimationFrame(timer);

  updateTimer();
}

function restartVideo() {
  pauseVideo();
  elapsed = 0;
  sceneStart = 0;
  showScene(0);
  updateTimer();
}

updateTimer();
</script>
